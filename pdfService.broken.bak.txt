// -------------------------------------------------------------
// Text sanitisation (WinAnsi safety)
// -------------------------------------------------------------

/**
 * Sanitises input text so pdf-lib can never throw a WinAnsi encoding error.
 * Common typographic / symbolic characters are transliterated to sensible ASCII
 * equivalents; anything still outside the WinAnsi range (emoji, CJK, ...) is
 * replaced with a literal '?' rather than being silently deleted.
 */
export function cleanTextForPdf(text: string | null | undefined): string {
  if (text == null) return '';
  let s = String(text);
  // Collapse control characters that pdf-lib cannot encode.
  s = s.replace(/\r\n?/g, ' ').replace(/[\n\t\f\v]/g, ' ');
  // Check markers (listed first so they survive the generic tick handling).
  s = s.replace(/\[\u2713\]/g, '[X]').replace(/\[\u2714\]/g, '[X]');
  s = s.replace(/[\u2713\u2714\u2611\u2612\u2705]/g, '[X]');
  s = s.replace(/[\u2610\u25A1\u25FB]/g, '[ ]');
  // Quotes / primes / guillemets
  s = s.replace(/[\u2018\u2019\u201A\u201B\u2032\u2035]/g, "'");
  s = s.replace(/[\u201C\u201D\u201E\u201F\u2033\u2036]/g, '"');
  s = s.replace(/[\u2039]/g, '<').replace(/[\u203A]/g, '>');
  s = s.replace(/[\u00AB]/g, '<<').replace(/[\u00BB]/g, '>>');
  // Dashes
  s = s.replace(/[\u2012\u2013\u2014\u2015\u2212]/g, '-');
  // Bullets / list markers
  s = s.replace(/[\u2022\u2023\u25AA\u25AB\u25CF\u25E6\u2043\u2219\u25FE\u25FC\u25C6]/g, '-');
  // Ellipsis
  s = s.replace(/\u2026/g, '...');
  // Fractions
  s = s.replace(/\u00BD/g, '1/2').replace(/\u00BC/g, '1/4').replace(/\u00BE/g, '3/4');
  s = s.replace(/\u2153/g, '1/3').replace(/\u2154/g, '2/3');
  // Fancy / non breaking spaces and misc symbols
  s = s.replace(/[\u00A0\u2007\u202F\u2009\u200A]/g, ' ');
  s = s.replace(/\u2122/g, '(TM)').replace(/\u00AE/g, '(R)').replace(/\u00A9/g, '(C)');
  s = s.replace(/\u20AC/g, 'EUR').replace(/\u00A3/g, 'GBP').replace(/\u00A5/g, 'JPY');
  // Anything left outside WinAnsi (emoji, CJK, ...) becomes a visible placeholder.
  return s.replace(/[^\x20-\x7E\xA0-\xFF]/g, '?');
}

// -------------------------------------------------------------
// Defensive prototype patches
// -------------------------------------------------------------
// pdf-lib throws when asked to encode a character outside the font's WinAnsi
// encoding. Rather than trusting every call site we sanitise centrally. The
// overrides are signature-safe so every pdf-lib overload (and `tsc`) is happy.
type AnyFunction = (...args: any[]) => any;

const originalDrawText = PDFPage.prototype.drawText as unknown as AnyFunction;
(PDFPage.prototype as unknown as { drawText: AnyFunction }).drawText = function (...args: any[]) {
  if (typeof args[0] === 'string') args[0] = cleanTextForPdf(args[0]);
  return originalDrawText.apply(this, args);
};

const originalWidthOfTextAtSize = PDFFont.prototype.widthOfTextAtSize as unknown as AnyFunction;
(PDFFont.prototype as unknown as { widthOfTextAtSize: AnyFunction }).widthOfTextAtSize = function (...args: any[]) {
  if (typeof args[0] === 'string') args[0] = cleanTextForPdf(args[0]);
  return originalWidthOfTextAtSize.apply(this, args);
};

/**
 * Utility to format currency values cleanly
 */
export function formatCurrency(amount: number | '' | undefined | null): string {
  if (amount === '' || amount === undefined || amount === null || isNaN(Number(amount))) {
    return '$0.00';
  }
  return new Intl.NumberFormat('en-US', {
    style: 'currency',
    currency: 'USD',
  }).format(Number(amount));
}

/**
 * Triggers a browser file download of PDF bytes
 */
export function downloadPdf(bytes: Uint8Array, filename: string) {
  const blob = new Blob([bytes as unknown as BlobPart], { type: 'application/pdf' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = filename;
  document.body.appendChild(a);
  a.click();
  document.body.removeChild(a);
  setTimeout(() => URL.revokeObjectURL(url), 10000);
}

/**
 * Creates an in-memory Object URL for in-app PDF previewing in an iframe/modal
 */
export function createPdfBlobUrl(bytes: Uint8Array): string {
  const blob = new Blob([bytes as unknown as BlobPart], { type: 'application/pdf' });
  return URL.createObjectURL(blob);
}

function drawHeaderLogo(
  page: PDFPage,
  fontBold: PDFFont,
  fontReg: PDFFont,
  branchInfo: RestorationJobData['branch']
) {
  // Red "H" mark
  page.drawRectangle({ x: 40, y: 730, width: 10, height: 32, color: COLOR_RED });
  page.drawRectangle({ x: 58, y: 730, width: 10, height: 32, color: COLOR_RED });
  page.drawRectangle({ x: 50, y: 742, width: 8, height: 8, color: COLOR_RED });

  // Logo text
  page.drawText('Hays + Sons', { x: 74, y: 746, size: 18, font: fontBold, color: COLOR_DARK });
  page.drawText('Your Disaster Recovery Professionals', {
    x: 75,
    y: 734,
    size: 7.5,
    font: fontReg,
    color: COLOR_GRAY,
  });

  // Branch contact block - genuinely right aligned, and every value guarded so
  // no `undefined` can ever reach the page.
  const rightX = CONTENT_RIGHT;
  const rightMinX = 300;
  const division = txt(branchInfo?.division, 'Hays and Sons - Fort Wayne');
  const addressLine = [val(branchInfo?.address), val(branchInfo?.cityStateZip)].filter(Boolean).join(', ');
  const phoneFaxBits: string[] = [];
  const phone = val(branchInfo?.phone);
  const fax = val(branchInfo?.fax);
  if (phone) phoneFaxBits.push(`Phone: ${phone}`);
  if (fax) phoneFaxBits.push(`Fax: ${fax}`);
  const contactLine = phoneFaxBits.join('  |  ');

  drawRightAligned(page, division, { rightX, minX: rightMinX, y: 756, size: 8.5, font: fontBold, color: COLOR_DARK, minSize: 6 });

    drawRightAligned(page, addressLine, { rightX, minX: rightMinX, y: 745, size: 7.5, font: fontReg, color: COLOR_GRAY, minSize: 5.5 });
  }
  if (contactLine) {
    drawRightAligned(page, contactLine, { rightX, minX: rightMinX, y: 734, size: 7.5, font: fontReg, color: COLOR_GRAY, minSize: 5.5 });
  }

  // Header separator line
  page.drawLine({ start: { x: MARGIN_X, y: HEADER_RULE_Y }, end: { x: CONTENT_RIGHT, y: HEADER_RULE_Y }, thickness: 1, color: COLOR_DARK });
}

interface DocFonts {
  pdfDoc: PDFDocument;
  font: PDFFont;
  fontBold: PDFFont;
  fontItalic: PDFFont;
}

/**
 * Creates a document pre-loaded with the three permitted standard fonts.
 * Every generator funnels through here so page geometry stays consistent.
 */
async function createDocument(): Promise<DocFonts> {
  const pdfDoc = await PDFDocument.create();
  const font = await pdfDoc.embedFont(StandardFonts.Helvetica);
  const fontBold = await pdfDoc.embedFont(StandardFonts.HelveticaBold);
  const fontItalic = await pdfDoc.embedFont(StandardFonts.HelveticaOblique);
  return { pdfDoc, font, fontBold, fontItalic };
}

/** Returns '' for null / undefined, otherwise the sanitised trimmed string. */
function val(v: unknown): string {
  if (v === null || v === undefined) return '';
  return cleanTextForPdf(String(v)).trim();
}

/** Like `val`, but substitutes `fallback` when the value is blank. */
function txt(v: unknown, fallback = 'N/A'): string {
  const s = val(v);
  return s.length > 0 ? s : fallback;
}

/** Measures a string, tolerating any encoding edge case. */
function measure(font: PDFFont, text: unknown, size: number): number {
  const s = cleanTextForPdf(val(text));
  if (!s) return 0;
  try {
    return font.widthOfTextAtSize(s, size);
  } catch {
    return 0;
  }
}

/** Hard character-level break for a token that is wider than the column. */
function breakToken(font: PDFFont, token: string, size: number, maxWidth: number): string[] {
  const chunks: string[] = [];
  let chunk = '';
  for (const ch of token) {
    const test = chunk + ch;
    if (chunk && measure(font, test, size) > maxWidth) {
      chunks.push(chunk);
      chunk = ch;
    } else {
      chunk = test;
    }
  }
  if (chunk) chunks.push(chunk);
  return chunks.length ? chunks : [token];
}

/**
 * Word-wraps text to `maxWidth`, honouring explicit newlines, and breaking
 * over-long tokens (emails / URLs) at character level. No line is ever wider
 * than `maxWidth`.
 */
function wrapText(font: PDFFont, text: unknown, size: number, maxWidth: number): string[] {
  if (text === null || text === undefined) return [];
  const raw = String(text);
  if (!raw.trim()) return [];

  const lines: string[] = [];
  for (const paragraph of raw.split(/\r?\n/)) {
    const words = cleanTextForPdf(paragraph).split(/\s+/).filter((w) => w.length > 0);
    if (words.length === 0) continue;

    let current = '';
    for (const word of words) {
      const fitsWithCurrent = current ? measure(font, `${current} ${word}`, size) <= maxWidth : measure(font, word, size) <= maxWidth;
      if (fitsWithCurrent) {
        current = current ? `${current} ${word}` : word;
        continue;
      }
      if (current) {
        lines.push(current);
        current = '';
      }
      if (measure(font, word, size) > maxWidth) {
        const chunks = breakToken(font, word, size, maxWidth);
        for (let i = 0; i < chunks.length - 1; i++) lines.push(chunks[i]);
        current = chunks[chunks.length - 1];
      } else {
        current = word;
      }
    }
    if (current) lines.push(current);
  }
  return lines;
}

/**
 * Shrinks text to fit `maxWidth` (down to `minSize`) and then ellipsises with
 * '…' so the returned text NEVER exceeds `maxWidth`.
 */
function fitText(font: PDFFont, text: unknown, size: number, maxWidth: number, minSize = 6): { text: string; size: number } {
  const clean = val(text);
  if (!clean || maxWidth <= 0) return { text: '', size };
  let s = size;
  while (s > minSize) {
    if (measure(font, clean, s) <= maxWidth) return { text: clean, size: s };
    s = Math.max(minSize, Math.round((s - 0.25) * 100) / 100);
    if (s === minSize) break;
  }
  if (measure(font, clean, s) <= maxWidth) return { text: clean, size: s };
  let out = clean;
  while (out.length > 0 && measure(font, out + ELLIPSIS, s) > maxWidth) {
    out = out.slice(0, -1);
  }
  return { text: out ? out + ELLIPSIS : '', size: s };
}

/** Draws text via `fitText`. Returns what was actually drawn. */
function drawTextFit(
  page: PDFPage,
  text: unknown,
  options: { x: number; y: number; size: number; font: PDFFont; color?: any; maxWidth: number; minSize?: number }
): { text: string; size: number } {
  const fitted = fitText(options.font, text, options.size, options.maxWidth, options.minSize);
  if (fitted.text) {
    page.drawText(fitted.text, { x: options.x, y: options.y, size: fitted.size, font: options.font, color: options.color ?? COLOR_DARK });
  }
  return fitted;
}

/**
 * Wraps + draws a paragraph. Returns the next `y` (already advanced past the
 * last drawn line). When `maxLines` is exceeded an ellipsis is appended to the
 * final drawn line so text is never silently dropped.
 */
function drawParagraph(
  page: PDFPage,
  text: unknown,
  options: {
    x: number;
    y: number;
    maxWidth: number;
    lineHeight: number;
    size: number;
    font: PDFFont;
    color?: any;
    maxLines?: number;
    ellipsis?: boolean;
  }
): number {
  const lines = wrapText(options.font, text, options.size, options.maxWidth);
  if (lines.length === 0) return options.y;

  const maxLines = options.maxLines ?? Number.POSITIVE_INFINITY;
  const drawCount = Math.max(0, Math.min(lines.length, maxLines));
  const useEllipsis = options.ellipsis ?? true;
  const color = options.color ?? COLOR_DARK;

  let y = options.y;
  for (let i = 0; i < drawCount; i++) {
    let line = lines[i];
    if (useEllipsis && i === drawCount - 1 && lines.length > drawCount) {
      while (line.length > 0 && measure(options.font, line + ELLIPSIS, options.size) > options.maxWidth) {
        line = line.slice(0, -1);
      }
      line += ELLIPSIS;
    }
    page.drawText(line, { x: options.x, y, size: options.size, font: options.font, color });
    y -= options.lineHeight;
  }
  return y;
}

/** Draws text right-aligned at `rightX`, shrunk to fit between `minX` and `rightX`. */
function drawRightAligned(
  page: PDFPage,
  text: unknown,
  options: { rightX: number; y: number; size: number; font: PDFFont; color?: any; minSize?: number; minX?: number }
): void {
  const minX = options.minX ?? MARGIN_X;
  const fitted = fitText(options.font, text, options.size, Math.max(0, options.rightX - minX), options.minSize);
  if (!fitted.text) return;
  const w = measure(options.font, fitted.text, fitted.size);
  page.drawText(fitted.text, {
    x: options.rightX - w,
    y: options.y,
    size: fitted.size,
    font: options.font,
    color: options.color ?? COLOR_DARK,
  });
}

/** Draws a label followed by an auto-fitted value, underlined to its measured end. */
function drawField(
  page: PDFPage,
  label: string,
  value: unknown,
  options: {
    x: number;
    y: number;
    maxWidth: number;
    labelSize?: number;
    valueSize?: number;
    labelWidth?: number;
    font: PDFFont;
    fontBold: PDFFont;
    color?: any;
    valueColor?: any;
    underline?: boolean;
    gap?: number;
  }
): void {
  const labelSize = options.labelSize ?? 8;
  const valueSize = options.valueSize ?? labelSize;
  const gap = options.gap ?? 6;
  page.drawText(cleanTextForPdf(label), {
    x: options.x,
    y: options.y,
    size: labelSize,
    font: options.fontBold,
    color: options.color ?? COLOR_DARK,
  });
  const labelW = options.labelWidth ?? measure(options.fontBold, label, labelSize);
  const valueX = options.x + labelW + gap;
  const availW = Math.max(0, options.maxWidth - labelW - gap);
  const fitted = drawTextFit(page, value, {
    x: valueX,
    y: options.y,
    size: valueSize,
    font,
    color: options.valueColor ?? options.color ?? COLOR_DARK,
    maxWidth: availW,
    minSize: 6,
  });
  if ((options.underline ?? true) && fitted.text) {
    const w = Math.min(measure(options.font, fitted.text, fitted.size), availW);
    page.drawLine({ start: { x: valueX, y: options.y - 2 }, end: { x: valueX + w, y: options.y - 2 }, thickness: 0.75, color: COLOR_LINE });
  }
}

/** Draws a label at `labelX` with a boxed, auto-fitted value starting at `boxX`. */
function drawBoxRow(
  page: PDFPage,
  font: PDFFont,
  fontBold: PDFFont,
  label: string,
  value: unknown,
  options: {
    labelX: number;
    boxX: number;
    boxW: number;
    y: number;
    labelSize?: number;
    valueSize?: number;
    boxH?: number;
    valueColor?: any;
  }
): void {
  const labelSize = options.labelSize ?? 8;
  const valueSize = options.valueSize ?? 8.5;
  const boxH = options.boxH ?? 16;
  page.drawText(cleanTextForPdf(label), { x: options.labelX, y: options.y, size: labelSize, font: fontBold, color: COLOR_DARK });
  page.drawRectangle({
    x: options.boxX,
    y: options.y - boxH + 2,
    width: options.boxW,
    height: boxH,
    borderColor: COLOR_BORDER,
    borderWidth: 0.75,
  });
  drawTextFit(page, value, {
    x: options.boxX + 4,
    y: options.y,
    size: valueSize,
    font,
    color: options.valueColor ?? COLOR_DARK,
    maxWidth: options.boxW - 8,
    minSize: 6,
  });
}

/** Consistent filled section-header bar with a left or centred title. */
function drawSectionHeader(
  page: PDFPage,
  fontBold: PDFFont,
  title: string,
  options: { x: number; y: number; width: number; fill?: any; height?: number; textSize?: number; align?: 'left' | 'center'; color?: any }
): void {
  const h = options.height ?? 15;
  page.drawRectangle({
    x: options.x,
    y: options.y - h,
    width: options.width,
    height: h,
    color: options.fill ?? COLOR_CYAN,
    borderColor: COLOR_LINE,
    borderWidth: 0.5,
  });
  const textSize = options.textSize ?? 8.5;
  const clean = cleanTextForPdf(title);
  const tw = measure(fontBold, clean, textSize);
  const tx = options.align === 'center' ? options.x + (options.width - tw) / 2 : options.x + 5;
  page.drawText(clean, { x: tx, y: options.y - h + 4.5, size: textSize, font: fontBold, color: options.color ?? COLOR_DARK });
}

/** Draws a single line of text horizontally centred on the page. */
function drawCenteredText(
  page: PDFPage,
  text: unknown,
  options: { y: number; size: number; font: PDFFont; color?: any; maxWidth?: number; minSize?: number }
): void {
  const maxWidth = options.maxWidth ?? CONTENT_W;
  const fitted = fitText(options.font, text, options.size, maxWidth, options.minSize);
  if (!fitted.text) return;
  const w = measure(options.font, fitted.text, fitted.size);
  page.drawText(fitted.text, {
    x: MARGIN_X + (maxWidth - w) / 2,
    y: options.y,
    size: fitted.size,
    font: options.font,
    color: options.color ?? COLOR_DARK,
  });
}

/**
 * Label on the left, amount right-aligned at `rightX`, with dot leaders that
 * always stop short of the amount so they can never collide.
 */
function drawLeaderRow(
  page: PDFPage,
  font: PDFFont,
  fontBold: PDFFont,
  label: string,
  amount: string,
  options: { x: number; rightX: number; y: number; size: number; color?: any; amountColor?: any }
): void {
  const amountW = measure(fontBold, amount, options.size);
  const amountX = options.rightX - amountW;
  const labelMax = Math.max(0, amountX - options.x - 24);
  const drawn = drawTextFit(page, label, {
    x: options.x,
    y: options.y,
    size: options.size,
    font,
    color: options.color ?? COLOR_DARK,
    maxWidth: labelMax,
    minSize: 6,
  });
  const dotStart = options.x + measure(font, drawn.text, drawn.size) + 5;
  const dotEnd = amountX - 6;
  if (dotEnd > dotStart) {
    const dotW = measure(font, '.', 7);
    if (dotW > 0) {
      const count = Math.floor((dotEnd - dotStart) / dotW);
      if (count > 0) page.drawText('.'.repeat(count), { x: dotStart, y: options.y, size: 7, font, color: COLOR_GRAY });
    }
  }
  page.drawText(amount, { x: amountX, y: options.y, size: options.size, font: fontBold, color: options.amountColor ?? COLOR_DARK });
}

/**
 * Branded footer. Left = brand, centre = document label + page indicator,
 * right = job number + customer. Long values are auto-fitted so the three
 * zones can never overlap.
 */
function drawFooter(
  page: PDFPage,
  font: PDFFont,
  data: RestorationJobData,
  options: { pageNum?: number; totalPages?: number; label?: string; ruleY?: number } = {}
): void {
  const ruleY = options.ruleY ?? FOOTER_RULE_Y;
  const textY = ruleY - 10;
  page.drawLine({ start: { x: MARGIN_X, y: ruleY }, end: { x: CONTENT_RIGHT, y: ruleY }, thickness: 0.5, color: COLOR_LINE });

  drawTextFit(page, 'Hays + Sons Complete Restoration \u2022 Fort Wayne Division', {
    x: MARGIN_X,
    y: textY,
    size: 6.5,
    font,
    color: COLOR_GRAY,
    maxWidth: 185,
    minSize: 5,
  });

  const jobNumber = val(data.customer?.jobNumber);
  const customerName = val(data.customer?.customerName);
  const rightBits: string[] = [];
  if (jobNumber) rightBits.push(`Job #${jobNumber}`);
  if (customerName) rightBits.push(customerName);
  if (rightBits.length) {
    drawRightAligned(page, rightBits.join(' \u2022 '), {
      rightX: CONTENT_RIGHT,
      minX: 400,
      y: textY,
      size: 6.5,
      font,
      color: COLOR_GRAY,
      minSize: 5,
    });
  }

  const centerBits: string[] = [];
  if (options.label) centerBits.push(options.label);
  if (options.pageNum && options.totalPages) centerBits.push(`Page ${options.pageNum} of ${options.totalPages}`);
  const centerText = centerBits.join('  |  ');
  if (centerText) {
    const fitted = fitText(font, centerText, 6.5, 150, 5);
    if (fitted.text) {
      const w = measure(font, fitted.text, fitted.size);
      page.drawText(fitted.text, { x: (PAGE_W - w) / 2, y: textY, size: fitted.size, font, color: COLOR_GRAY });
    }
  }
}

/** Standard signature line with an optional printed name beneath it. */
function drawSignatureLine(
  page: PDFPage,
  options: {
    x: number;
    y: number;
    width: number;
    label: string;
    printedName?: string;
    font: PDFFont;
    fontBold: PDFFont;
    labelSize?: number;
    nameSize?: number;
  }
): void {
  page.drawLine({ start: { x: options.x, y: options.y }, end: { x: options.x + options.width, y: options.y }, thickness: 1, color: COLOR_DARK });
  drawTextFit(page, options.label, {
    x: options.x,
    y: options.y - 11,
    size: options.labelSize ?? 8,
    font: options.fontBold,
    color: COLOR_DARK,
    maxWidth: options.width,
    minSize: 6,
  });
  if (options.printedName) {
    drawTextFit(page, `Printed: ${val(options.printedName)}`, {
      x: options.x,
      y: options.y - 21,
      size: options.nameSize ?? 7.5,
      font: options.font,
      color: COLOR_GRAY,
      maxWidth: options.width,
      minSize: 6,
    });
  }
}

/** Crisp square checkbox with a green X when checked. */
function drawCheckbox(page: PDFPage, options: { x: number; y: number; checked: boolean; size?: number; color?: any; borderColor?: any }): void {
  const s = options.size ?? 9;
  page.drawRectangle({
    x: options.x,
    y: options.y,
    width: s,
    height: s,
    borderColor: options.borderColor ?? COLOR_DARK,
    borderWidth: 0.9,
    color: COLOR_WHITE,
  });
  if (options.checked) {
    const pad = Math.max(1.5, s * 0.22);
    const c = options.color ?? COLOR_GREEN;
    page.drawLine({ start: { x: options.x + pad, y: options.y + pad }, end: { x: options.x + s - pad, y: options.y + s - pad }, thickness: 1.1, color: c });
    page.drawLine({ start: { x: options.x + pad, y: options.y + s - pad }, end: { x: options.x + s - pad, y: options.y + pad }, thickness: 1.1, color: c });
  }
}

export async function generateCancellationNotice(data: RestorationJobData): Promise<Uint8Array> {
  const { pdfDoc, font, fontBold } = await createDocument();
  const page = pdfDoc.addPage([PAGE_W, PAGE_H]);

  drawHeaderLogo(page, fontBold, font, data.branch);

  let y = 690;
  drawCenteredText(page, 'NOTICE OF CANCELLATION', { y, size: 14, font: fontBold });

  y -= 24;
  drawTextFit(page, `Job #: ${txt(data.customer.jobNumber)}   |   Customer: ${txt(data.customer.customerName)}`, {
    x: 50, y, size: 8.5, font: fontBold, color: COLOR_RED, maxWidth: 340,
  });
  drawTextFit(page, `Agreement Date: ${txt(data.insurance.dateReceived, new Date().toLocaleDateString('en-US'))}`, {
    x: 400, y, size: 8, font: fontBold, maxWidth: 172,
  });
  y -= 14;
  drawTextFit(page, `Property Address: ${txt(data.customer.lossAddress)}`, { x: 50, y, size: 8, font, maxWidth: 522 });

  const paragraph = (text: string, opts: { indent?: number; size?: number; font?: PDFFont } = {}) => {
    const indent = opts.indent ?? 0;
    y -= 11.5;
    y = drawParagraph(page, text, {
      x: 50 + indent,
      y,
      maxWidth: 522 - indent,
      lineHeight: 11.5,
      size: opts.size ?? 8,
      font: opts.font ?? font,
      color: COLOR_DARK,
    }) + 11.5;
  };

  y -= 12;
  paragraph(
    `You, the OWNER, may cancel this AGREEMENT by mailing, delivering, or submitting by electronic mail a signed and dated copy of this cancellation notice to the CONTRACTOR, Hays & Sons Construction, Inc., at ${txt(data.branch.address)}, ${txt(data.branch.cityStateZip)}, or ${txt(data.branch.managerEmail)}, at any time before midnight on the third business day after the later of:`
  );

  paragraph('(A) The date of this AGREEMENT is signed by you and the CONTRACTOR, Hays & Sons Construction, Inc.', { indent: 25, font: fontBold });
  paragraph(
    '(B) If applicable, the date you receive written notification from your insurance company of a final determination as to whether all or any part of your claim or this AGREEMENT is a covered loss under your policy.',
    { indent: 25, font: fontBold }
  );

  paragraph(
    'If you cancel this AGREEMENT, any payment made by you under the AGREEMENT will be returned within ten (10) business days following receipt by the CONTRACTOR of your cancellation notice, minus any amounts you may owe for work already done by the CONTRACTOR, Hays & Sons Construction, Inc.'
  );
  paragraph(
    'If you cancel, you must make available to CONTRACTOR, at your property and in substantially as good condition as when received, any goods delivered to you under this contract of sale; or you may comply with instructions of CONTRACTOR regarding return shipment.'
  );

  y -= 25;
  drawCenteredText(page, '*************************************', { y, size: 10, font: fontBold });

  y -= 25;
  drawTextFit(page, 'I HEREBY CANCEL THIS AGREEMENT.', { x: 50, y, size: 9, font: fontBold, maxWidth: 522 });

  y -= 45;
  drawSignatureLine(page, { x: 50, y, width: 300, label: 'Owner\'s Signature', font, fontBold });
  drawSignatureLine(page, { x: 400, y, width: 172, label: 'Date', font, fontBold });

  y -= 32;
  drawTextFit(page, `Printed Name:        ${txt(data.customer.customerName)}`, { x: 50, y, size: 8.5, font: fontBold, maxWidth: 522 });

  drawFooter(page, font, data, { pageNum: 1, totalPages: 1, label: 'Notice of Cancellation' });

  return pdfDoc.save();

// 8. UNIFIED COMPLETE 8-PAGE PACKET GENERATOR
// -------------------------------------------------------------
export async function generateCompletePacket(data: RestorationJobData): Promise<Uint8Array> {
  const mergedPdf = await PDFDocument.create();

  // Generate each document in sequential order
  const [
    prelimBytes,
    welcomeBytes,
    mortgageBytes,
    contractBytes,
    cancellationBytes,
    changeOrderBytes,
    checklistBytes,
  ] = await Promise.all([
    generatePreliminaryReport(data),
    generateWelcomeLetter(data),
    generateMortgageAuth(data),
    generateContract(data), // 2 pages
    generateCancellationNotice(data),
    generateChangeOrder(data),
    generateProductionChecklist(data),
  ]);

  const docs = await Promise.all([
    PDFDocument.load(prelimBytes),
    PDFDocument.load(welcomeBytes),
    PDFDocument.load(mortgageBytes),
    PDFDocument.load(contractBytes),
    PDFDocument.load(cancellationBytes),
    PDFDocument.load(changeOrderBytes),
    PDFDocument.load(checklistBytes),
  ]);

  for (const doc of docs) {
    const copiedPages = await mergedPdf.copyPages(doc, doc.getPageIndices());
    copiedPages.forEach((page) => mergedPdf.addPage(page));
  }

  // Stamp running page footers across all pages for complete packet integrity
  const fontFooter = await mergedPdf.embedFont(StandardFonts.Helvetica);
  const totalPages = mergedPdf.getPageCount();
  for (let i = 0; i < totalPages; i++) {
    const p = mergedPdf.getPage(i);
    p.drawText(
      `Page ${i + 1} of ${totalPages}  |  Hays + Sons Complete Restoration  |  Job #${data.customer.jobNumber}  |  ${data.customer.customerName}`,
      {
        x: 45,
        y: 16,
        size: 7,
        font: fontFooter,
        color: COLOR_GRAY,
      }
    );
  }

  return mergedPdf.save();
}

// Form Template Filling Demonstration Function (AcroForms with pdf-lib)
// -------------------------------------------------------------
export async function fillPdfFormTemplate(
  templatePdfBytes: Uint8Array,
  data: RestorationJobData
): Promise<Uint8Array> {
  const pdfDoc = await PDFDocument.load(templatePdfBytes);
  const form = pdfDoc.getForm();

  const trySetText = (fieldName: string, value: string) => {
    try {
      const field = form.getTextField(fieldName);
      if (field) field.setText(value);
    } catch {
      // field not found, ignore
    }
  };

  const trySetCheck = (fieldName: string, isChecked: boolean) => {
    try {
      const field = form.getCheckBox(fieldName);
      if (field) {
        if (isChecked) field.check();
        else field.uncheck();
      }
    } catch {
      // field not found, ignore
    }
  };

  trySetText('JobNumber', data.customer.jobNumber);
  trySetText('CustomerName', data.customer.customerName);
  trySetText('MailingAddress', data.customer.mailingAddress);
  trySetText('LossAddress', data.customer.lossAddress);
  trySetText('Phone', data.customer.mobilePhone || data.customer.mainPhone);
  trySetText('Email', data.customer.email);
  trySetText('InsuranceCarrier', data.insurance.carrier);
  trySetText('ClaimNumber', data.insurance.claimNumber);
  trySetText('AdjusterName', data.insurance.primaryAdjuster);
  trySetText('TotalRCV', formatCurrency(data.financials.totalApprovedRcv));
  trySetText('Deductible', formatCurrency(data.financials.deductible));
  trySetText('NetClaimValue', formatCurrency(data.financials.netClaimValue));
  trySetText('DownPayment', formatCurrency(data.financials.downPayment));
  trySetText('MidProgressPayment', formatCurrency(data.financials.midProgressPayment));
  trySetText('Balance', formatCurrency(data.financials.balancePayment));
  trySetText('CommenceDays', String(data.financials.commenceDays));
  trySetText('CompleteDays', String(data.financials.completeDays));
  trySetCheck('MortgageOnClaim', data.mortgage.hasMortgage);
  trySetText('MortgageCompany', data.mortgage.mortgageCompany);
  trySetText('LoanNumber', data.mortgage.loanNumber);

  return pdfDoc.save();
}
